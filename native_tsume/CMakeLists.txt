cmake_minimum_required(VERSION 3.15)
project(tsume_native LANGUAGES CXX)
add_library(tsume SHARED tsume.cpp)
target_compile_features(tsume PRIVATE cxx_std_17)
if(MSVC)
  set_property(TARGET tsume PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(tsume PRIVATE /W4 /utf-8)
else()
  target_compile_options(tsume PRIVATE -Wall -Wextra)
endif()
